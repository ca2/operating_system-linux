cmake_minimum_required(VERSION 3.21)
project(cefclient LANGUAGES C CXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR CMAKE_CROSSCOMPILING
   OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
  message(FATAL_ERROR "This CEF wrapper project requires native Linux x86_64")
endif()
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "CEF configuration" FORCE)
endif()
set_property(CACHE CMAKE_BUILD_TYPE PROPERTY STRINGS Debug Release)
if(NOT CMAKE_BUILD_TYPE MATCHES "^(Debug|Release)$")
  message(FATAL_ERROR "Choose CMAKE_BUILD_TYPE=Debug or Release to match the prebuilt SDK")
endif()
get_filename_component(_workspace_parent "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(CEF_STORAGE_ROOT "${_workspace_parent}/operating_system/storage-linux" CACHE PATH "CEF staging directory")
if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
  set(CMAKE_INSTALL_PREFIX "${CEF_STORAGE_ROOT}" CACHE PATH "CEF staging directory" FORCE)
endif()
set(_sdk "$ENV{CEF_ROOT}")
if(NOT _sdk AND EXISTS "${CEF_STORAGE_ROOT}/x64/Release/cef-manifest.json")
  file(STRINGS "${CEF_STORAGE_ROOT}/x64/Release/cef-manifest.json" _line REGEX "\"sdk\":")
  string(REGEX REPLACE "^[^:]*: *\"([^\"]*)\".*$" "\\1" _sdk "${_line}")
endif()
set(CEF_ROOT "${_sdk}" CACHE PATH "Complete prebuilt CEF Standard Distribution")
if(NOT EXISTS "${CEF_ROOT}/cmake/FindCEF.cmake")
  message(FATAL_ERROR "Run _tool/cefclient/setup.sh first or provide -DCEF_ROOT=/path/to/cef_binary_VERSION_linux64")
endif()
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)

# Build the SDK's real static adapter. libcef.so itself is prebuilt.
add_custom_target(cefclient DEPENDS libcef_dll_wrapper)
set(_destination "x64/${CMAKE_BUILD_TYPE}")
install(TARGETS libcef_dll_wrapper ARCHIVE DESTINATION "${_destination}/library")
install(DIRECTORY "${CEF_ROOT}/${CMAKE_BUILD_TYPE}/" "${CEF_ROOT}/Resources/"
  DESTINATION "${_destination}/binary")
install(DIRECTORY "${CEF_ROOT}/include" DESTINATION include/cef)
install(FILES "${CEF_ROOT}/LICENSE.txt" "${CEF_ROOT}/README.txt" DESTINATION "${_destination}/cef")
install(CODE "
  set(_link \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/${_destination}/library/libcef.so\")
  if(IS_SYMLINK \"\${_link}\")
    file(REMOVE \"\${_link}\")
  endif()
  if(NOT EXISTS \"\${_link}\")
    file(CREATE_LINK ../binary/libcef.so \"\${_link}\" SYMBOLIC)
  endif()
")
